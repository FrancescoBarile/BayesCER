Simulation study

We compare the empirical Bayes and Gibbs sampling approaches implemented in BayesCER. The simulation results are precomputed and included with the package, so that the simulations do not need to be rerun when building the vignette.

The performance of the two approaches is assessed in terms of the estimation error for the representative graph and the scale parameter, as well as the accuracy of graph-level summary statistics.

ggplot(
  sim_data_result,
  aes(
    x = as.factor(n),
    y = discrepancy_C_hat_norm,
    fill = method
  )
) +
  geom_boxplot(
    aes(colour = after_scale(colorspace::darken(fill, 0.2))),
    position = position_dodge(width = 0.8),
    outlier.shape = NA,
    linewidth = 0.3,
    show.legend = c(colour = FALSE, fill = TRUE)
  ) +
  stat_boxplot(
    aes(colour = method),
    geom = "errorbar",
    position = position_dodge(width = 0.8),
    width = 0.2,
    show.legend = FALSE
  ) +
  stat_summary(
    fun = median,
    geom = "point",
    position = position_dodge(width = 0.8),
    size = 1.2,
    colour = "black",
    show.legend = FALSE
  ) +
  labs(
    x = "Sample size n",
    y = "",
    fill = "Method",
     title = bquote(
    "Normalized Hamming error " ~ d[H](C, hat(C))/M ~
    " for " ~ N == .(N_vec) ~ " and " ~ alpha == .(alpha_vec)
  )
    # title = paste0(
    #   expression("Normalized Hamming error " ~ d[H](C, hat(C))/M), 
    #   "for N = ", N_vec, "and ", expression(alpha), " = ",  alpha_vec)
  ) +
  theme_bw() +
  theme(
    panel.spacing.x = unit(0, "mm"),
    plot.margin = margin(rep(1, 4), unit = "mm"),
    strip.text = element_text(size = 14),
    axis.text = element_text(size = 14),
    strip.text.y.right = element_text(size = 14),
    strip.text.x.top = element_text(size = 14),
    axis.title.x = element_text(size = 14)
  )
